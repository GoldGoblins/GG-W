cmake_minimum_required(VERSION 3.16)
project(gg_presentation LANGUAGES C CXX)
file(GLOB LUA_SOURCE CONFIGURE_DEPENDS lua/*.c)
list(FILTER LUA_SOURCE EXCLUDE REGEX "/(linit|ldblib|liolib|loslib|loadlib)\\.c$")
add_library(gg_presentation SHARED gg_presentation.cpp ${LUA_SOURCE})
target_include_directories(gg_presentation PRIVATE lua)
target_compile_features(gg_presentation PRIVATE cxx_std_17)
set_target_properties(gg_presentation PROPERTIES C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)
target_link_libraries(gg_presentation PRIVATE m dl)
add_executable(gg_presentation_probe presentation_probe.cpp)
target_link_libraries(gg_presentation_probe PRIVATE gg_presentation)
enable_testing()
add_test(NAME presentation_probe COMMAND gg_presentation_probe ${CMAKE_CURRENT_SOURCE_DIR}/../../qml/game/lua/host.lua)
