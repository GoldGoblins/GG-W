cmake_minimum_required(VERSION 3.16)
project(gg_world LANGUAGES CXX)
add_library(gg_world SHARED gg_world.cpp)
target_compile_features(gg_world PUBLIC cxx_std_17)
target_compile_options(gg_world PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden)
target_include_directories(gg_world PUBLIC .)
add_executable(gg_world_probe world_probe.cpp)
target_link_libraries(gg_world_probe PRIVATE gg_world)
enable_testing()
add_test(NAME gg_world_probe COMMAND gg_world_probe)
