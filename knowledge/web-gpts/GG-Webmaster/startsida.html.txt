document.addEventListener('DOMContentLoaded', function(){

    const slides = [
        {
            image: 'https://www.goldgoblins.se/wp-content/uploads/2026/06/silverhange-med-onyx-46ct-sammet-1.webp',
            text: 'HÄNGSMYCKEN',
            link: '/hangsmycke'
        },
        {
            image: 'https://www.goldgoblins.se/wp-content/uploads/2026/06/hamrad-silverring-830-silver-produktbild-1.webp',
            text: 'RINGAR',
            link: '/ringar'
        },
        {
            image: 'https://www.goldgoblins.se/wp-content/uploads/2026/06/aurora-silverring-830-silver-produktbild-1.webp',
            text: 'RINGAR',
            link: '/ringar'
        }
    ];

    const bg1 = document.querySelector('.gg-bg-1');
    const bg2 = document.querySelector('.gg-bg-2');

    const button = document.getElementById('gg-heritage-btn');
    const text1 = document.querySelector('.gg-btn-text-1');
    const text2 = document.querySelector('.gg-btn-text-2');

    let current = 0;
    let showingFirstBg = true;
    let showingFirstText = true;

    bg1.style.backgroundImage = `url('${slides[0].image}')`;
    button.href = slides[0].link;
    text1.textContent = slides[0].text;

    setInterval(() => {

        current = (current + 1) % slides.length;

        if(showingFirstBg){
            bg2.style.backgroundImage = `url('${slides[current].image}')`;

            requestAnimationFrame(() => {
                bg2.style.opacity = '1';
                bg1.style.opacity = '0';
            });

        } else {
            bg1.style.backgroundImage = `url('${slides[current].image}')`;

            requestAnimationFrame(() => {
                bg1.style.opacity = '1';
                bg2.style.opacity = '0';
            });
        }

        if(showingFirstText){
            text2.textContent = slides[current].text;
            text2.style.opacity = '1';
            text1.style.opacity = '0';
        } else {
            text1.textContent = slides[current].text;
            text1.style.opacity = '1';
            text2.style.opacity = '0';
        }

        button.href = slides[current].link;

        showingFirstBg = !showingFirstBg;
        showingFirstText = !showingFirstText;

    }, 7777);

});
</script>
